async function askOverviewAI(){
  const input = document.getElementById('overviewAiPrompt');
  const output = document.getElementById('overviewAiOutput');
  if(!input || !output) return;

  const prompt = input.value.trim();
  if(!prompt){
    output.textContent = 'Type a question first.';
    output.style.color = 'var(--amber)';
    return;
  }

  output.style.color = 'var(--text2)';
  output.textContent = 'Querying the local model...';

  try {
    const reply = await window.VOLARIX_LLM.chat(`Use the context of this options dashboard and answer briefly: ${prompt}` , { temperature: 0.3 });
    output.textContent = reply;
    output.style.color = 'var(--text)';
  } catch (error) {
    output.textContent = 'AI request failed: ' + (error && error.message ? error.message : 'unknown error');
    output.style.color = 'var(--red)';
  }
}

function buildCalcContent(sym, strategy, strike, premium, dte) {
  const underlying = MKT.prices[sym] || strike;
  const contracts  = 1;
  const mult       = 100;

  // P&L function per strategy
  function pnlAt(price) {
    switch(strategy) {
      case 'Long Call':      return Math.max(-premium, price - strike - premium) * mult;
      case 'Long Put':       return Math.max(-premium, strike - price - premium) * mult;
      case 'Cash Secured Put': return Math.min(premium, premium - Math.max(0, strike - price)) * mult;
      case 'Covered Call':   return Math.min(premium, premium - Math.max(0, price - strike)) * mult;
      case 'Iron Condor': {
        const lowerPut = strike * 0.95, upperCall = strike * 1.05;
        const credit = premium;
        const putLoss  = Math.max(0, lowerPut - price);
        const callLoss = Math.max(0, price - upperCall);
        return (credit - putLoss - callLoss) * mult;
      }
      default: return (price - strike - premium) * mult;
    }
  }

  // Compute key metrics
  const prices     = Array.from({length:60}, (_,i) => underlying * (0.70 + i * 0.01));
  const pnls       = prices.map(p => pnlAt(p));
  const maxProfit  = Math.max(...pnls);
  const maxLoss    = Math.min(...pnls);
  const breakevens = [];
  for(let i=1; i<pnls.length; i++) {
    if((pnls[i-1] < 0 && pnls[i] >= 0) || (pnls[i-1] >= 0 && pnls[i] < 0)) {
      breakevens.push(prices[i].toFixed(2));
    }
  }
  const probProfit = (pnls.filter(p => p > 0).length / pnls.length * 100).toFixed(0);

  // Theta burn table (7 rows)
  const thetaBurnPerDay = premium * 0.03;
  const thetaRows = Array.from({length:7}, (_,i) => {
    const daysLeft = dte - (i+1)*Math.floor(dte/7);
    const remaining = (premium - thetaBurnPerDay * (i+1)*Math.floor(dte/7)).toFixed(2);
    return { day: (i+1)*Math.floor(dte/7), remaining: Math.max(0, remaining) };
  });

  $('calcContent').innerHTML = `
    <!-- Stats row -->
    <div style="display:grid;grid-template-columns:repeat(4,1fr);gap:10px;margin-bottom:16px">
      ${[
        ['Max Profit', maxProfit===Infinity?'Unlimited':'$'+maxProfit.toFixed(0), maxProfit>0?'green':'red'],
        ['Max Loss',   '$'+Math.abs(maxLoss).toFixed(0), 'red'],
        ['Breakeven',  breakevens.length ? '$'+breakevens[0] : 'N/A', 'amber'],
        ['Prob. Profit', probProfit+'%', +probProfit>50?'cyan':'amber'],
      ].map(([l,v,c]) => `<div class="calc-stat"><div class="calc-stat-val clr-${c}">${v}</div><div class="calc-stat-lbl">${l}</div></div>`).join('')}
    </div>

    <!-- Price slider -->
    <div style="margin-bottom:14px">
      <div style="display:flex;justify-content:space-between;font-size:11px;color:var(--text3);margin-bottom:6px">
        <span>Underlying Price</span>
        <span id="calcSliderVal" style="font-family:var(--mono);font-weight:700;color:var(--text)">$${underlying.toFixed(2)}</span>
      </div>
      <div class="slider-wrap">
        <input type="range" id="calcSlider"
          min="${(underlying*0.70).toFixed(0)}"
          max="${(underlying*1.30).toFixed(0)}"
          value="${underlying.toFixed(0)}"
          oninput="updateCalcSlider(this.value, ${strike}, ${premium}, '${strategy}')">
      </div>
      <div style="display:flex;justify-content:space-between;font-size:10px;color:var(--text3);margin-top:3px">
        <span>−30%</span><span style="color:var(--amber)">Current</span><span>+30%</span>
      </div>
    </div>

    <!-- P&L at current price -->
    <div id="calcCurrentPnl" style="text-align:center;padding:12px;background:var(--bg3);border-radius:var(--r);margin-bottom:14px;font-family:var(--mono);font-size:20px;font-weight:700;color:var(--cyan)">
      P&L at $${underlying.toFixed(2)}: ${pnlAt(underlying)>=0?'+':''}$${pnlAt(underlying).toFixed(0)}
    </div>

    <!-- P&L Canvas chart -->
    <div style="margin-bottom:14px">
      <div style="font-size:10px;font-weight:700;text-transform:uppercase;letter-spacing:.1em;color:var(--text3);margin-bottom:8px">P&L at Expiry</div>
      <div style="background:var(--bg3);border-radius:var(--r);overflow:hidden">
        <canvas id="calcChart" style="width:100%;height:120px;display:block"></canvas>
      </div>
    </div>

    <!-- Theta burn table -->
    <div style="margin-bottom:14px">
      <div style="font-size:10px;font-weight:700;text-transform:uppercase;letter-spacing:.1em;color:var(--text3);margin-bottom:8px">Theta Decay Projection</div>
      <div style="display:grid;grid-template-columns:repeat(7,1fr);gap:4px">
        ${thetaRows.map(r => `<div style="text-align:center;padding:8px 4px;background:var(--bg3);border-radius:var(--r)">

          <div style="font-size:9px;color:var(--text3);margin-top:3px">${r.day}d</div>
        </div>`).join('')}
      </div>
    </div>

    <div style="display:flex;gap:10px">
      <button class="btn btn-primary" style="flex:1;justify-content:center" onclick="closeCalc();showTab('paper')">Open Paper Trade →</button>

  `;

}

function updateCalcSlider(price, strike, premium, strategy) {
  price = +price;
  $('calcSliderVal').textContent = '$' + price.toFixed(2);
  function pnlAt(p) {
    switch(strategy) {
      case 'Long Call':      return Math.max(-premium, p - strike - premium) * 100;
      case 'Long Put':       return Math.max(-premium, strike - p - premium) * 100;
      case 'Cash Secured Put': return Math.min(premium, premium - Math.max(0, strike - p)) * 100;
      default: return (p - strike - premium) * 100;
    }
  }
  const pnl = pnlAt(price);
  const el  = $('calcCurrentPnl');
  if(el) {
    el.textContent = 'P&L at $' + price.toFixed(2) + ': ' + (pnl>=0?'+':'') + '$' + pnl.toFixed(0);
    el.style.color = pnl >= 0 ? 'var(--cyan)' : 'var(--red)';
  }
}

function drawCalcChart(sym, strategy, strike, premium) {
  const canvas = $('calcChart'); if(!canvas) return;
  const ctx = canvas.getContext('2d');
  const W = canvas.offsetWidth, H = canvas.offsetHeight;
  canvas.width = W; canvas.height = H;
  const underlying = MKT.prices[sym] || strike;
  const prices = Array.from({length:60}, (_,i) => underlying * (0.70 + i*0.01));
  function pnlAt(p) {
    switch(strategy) {
      case 'Long Call': return Math.max(-premium, p-strike-premium)*100;
      case 'Long Put':  return Math.max(-premium, strike-p-premium)*100;
      case 'Cash Secured Put': return Math.min(premium, premium-Math.max(0,strike-p))*100;
      default: return (p-strike-premium)*100;
    }
  }
  const pnls = prices.map(p => pnlAt(p));
  const minP = Math.min(...pnls), maxP = Math.max(...pnls);
  const range = (maxP - minP) || 1;
  const sy = v => (H-12)*(1-(v-minP)/range)+6;
  ctx.clearRect(0,0,W,H);
  // Zero line
  ctx.strokeStyle='rgba(255,255,255,0.15)'; ctx.lineWidth=1; ctx.setLineDash([3,3]);
  ctx.beginPath(); ctx.moveTo(0,sy(0)); ctx.lineTo(W,sy(0)); ctx.stroke(); ctx.setLineDash([]);
  // Fill
  const grad = ctx.createLinearGradient(0,0,0,H);
  grad.addColorStop(0,'rgba(0,212,170,0.3)'); grad.addColorStop(1,'rgba(0,212,170,0)');
  ctx.beginPath(); ctx.moveTo(0,sy(pnls[0]));
  pnls.forEach((v,i) => ctx.lineTo((i/(pnls.length-1))*W, sy(v)));
  ctx.lineTo(W,sy(0)); ctx.lineTo(0,sy(0)); ctx.closePath();
  ctx.fillStyle=grad; ctx.fill();
  // Line
  ctx.strokeStyle='#00D4AA'; ctx.lineWidth=2;
  ctx.beginPath(); pnls.forEach((v,i) => { const x=(i/(pnls.length-1))*W; i===0?ctx.moveTo(x,sy(v)):ctx.lineTo(x,sy(v)); }); ctx.stroke();
  // Current price marker
  const ci = Math.floor((underlying - prices[0]) / (prices[prices.length-1]-prices[0]) * (prices.length-1));
  if(ci>=0 && ci<pnls.length) {
    const cx=(ci/(pnls.length-1))*W, cy=sy(pnls[ci]);
    ctx.fillStyle='#FFB830'; ctx.beginPath(); ctx.arc(cx,cy,5,0,Math.PI*2); ctx.fill();
  }
}

const LS_KEY = 'volarix_state';

function saveState() {
  try {
    const toSave = {
      watchlist:  STATE.watchlist,
      theme:      STATE.theme,
      thresholds: STATE.thresholds,
      paperTrades:STATE.paperTrades,
    };
    localStorage.setItem(LS_KEY, JSON.stringify(toSave));
  } catch(e) { console.warn('saveState failed:', e); }
}

function loadState() {
  try {
    const raw = localStorage.getItem(LS_KEY);
    if(!raw) return;
    const saved = JSON.parse(raw);
    if(saved.watchlist  && Array.isArray(saved.watchlist))  STATE.watchlist  = saved.watchlist;
    if(saved.theme)                                          STATE.theme      = saved.theme;
    if(saved.thresholds && typeof saved.thresholds==='object') STATE.thresholds = Object.assign(STATE.thresholds, saved.thresholds);
    if(saved.paperTrades && Array.isArray(saved.paperTrades)) STATE.paperTrades = saved.paperTrades;
    // Apply theme immediately
    if(STATE.theme === 'light') {
      document.documentElement.setAttribute('data-theme','light');
      const btn = $('themeBtn'); if(btn) btn.textContent = '☀️';
    }
  } catch(e) { console.warn('loadState failed:', e); }
}

function buildJournalStats() {
  const trades = STATE.paperTrades;
  if(!trades.length) {
    set('tab-journal','<div class="card card-body" style="text-align:center;padding:40px;color:var(--text3)"><div style="font-size:28px;margin-bottom:10px">📋</div><div>No closed trades yet. Run paper trades to see statistics.</div></div>');
    return;
  }
  const wins     = trades.filter(t => t.cls==='win');
  const losses   = trades.filter(t => t.cls==='loss');
  const winRate  = (wins.length/trades.length*100).toFixed(0);
  const byStrat  = {};
  trades.forEach(t => {
    if(!byStrat[t.s]) byStrat[t.s] = {wins:0,losses:0};
    t.cls==='win' ? byStrat[t.s].wins++ : byStrat[t.s].losses++;
  });
  const pnlNums  = trades.map(t => {
    const n = parseFloat(t.pnl.replace(/[^0-9.-]/g,''));
    return t.cls==='win' ? n : -n;
  });
  const bestTrade  = trades.reduce((a,b) => {
    const av = parseFloat(a.pnl.replace(/[^0-9.-]/g,'')), bv = parseFloat(b.pnl.replace(/[^0-9.-]/g,''));
    return av > bv ? a : b;
  });
  const worstTrade = trades.reduce((a,b) => {
    const av = parseFloat(a.pnl.replace(/[^0-9.-]/g,'')), bv = parseFloat(b.pnl.replace(/[^0-9.-]/g,''));
    return av < bv ? a : b;
  });
  const totalPnl = pnlNums.reduce((s,v) => s+v, 0);

  set('tab-journal', `
    <div style="font-size:15px;font-weight:700;margin-bottom:4px">📊 Trade Journal Statistics</div>
    <div style="font-size:11px;color:var(--text3);margin-bottom:16px">Calculated from ${trades.length} closed paper trades</div>

      ${[
        ['Win Rate', winRate+'%', +winRate>=60?'green':+winRate>=40?'amber':'red'],
        ['Total P&L', (totalPnl>=0?'+':'')+'$'+Math.abs(totalPnl).toFixed(0), totalPnl>=0?'cyan':'red'],
        ['Total Trades', trades.length, 'text'],
        ['Wins / Losses', wins.length+' / '+losses.length, 'text'],
      ].map(([l,v,c]) => `<div class="journal-stat"><div class="journal-stat-val clr-${c}">${v}</div><div class="journal-stat-lbl">${l}</div></div>`).join('')}
    </div>
    <div style="display:grid;grid-template-columns:1fr 1fr;gap:14px;margin-bottom:14px">
      <div class="card card-body">
        <div style="font-size:12px;font-weight:700;margin-bottom:10px">Win Rate by Strategy</div>
        ${Object.entries(byStrat).map(([strat,data]) => {
          const total = data.wins+data.losses;

// P5 — IV RANK HISTORICAL CHART
// ══════════════════════════════════════════
function drawIVHistory(canvasId, sym) {
  const canvas = $(canvasId); if(!canvas) return;
  const ctx = canvas.getContext('2d');
  const W = canvas.offsetWidth, H = canvas.offsetHeight;
  canvas.width=W; canvas.height=H;
  // Simulate 52 weekly IV readings
  const baseIV = (FUNDAMENTALS[sym]||{}).earningsDate ? 60 : 30;
  const ivHistory = Array.from({length:52}, (_,i) => {
    const trend = Math.sin(i/8)*15;
    return Math.max(10, Math.min(150, baseIV + trend + (Math.random()-0.5)*20));
  });
  const currentIV = ivHistory[ivHistory.length-1];
  const maxIV = Math.max(...ivHistory), minIV = Math.min(...ivHistory);
  const range = (maxIV - minIV) || 1;
  const sy = v => (H-8)*(1-(v-minIV)/range)+4;
  const bW = W/52;
  ctx.clearRect(0,0,W,H);
  ivHistory.forEach((iv,i) => {
    const pct = (iv-minIV)/range;
    const r = Math.round(255*pct), g = Math.round(212*(1-pct));
    ctx.fillStyle = `rgba(${r},${g},100,0.7)`;
    ctx.fillRect(i*bW+1, sy(iv), bW-2, H-sy(iv)-4);
  });
  // Current IV marker line
  ctx.strokeStyle='var(--amber)'; ctx.lineWidth=1.5; ctx.setLineDash([3,2]);
  ctx.beginPath(); ctx.moveTo(0,sy(currentIV)); ctx.lineTo(W,sy(currentIV)); ctx.stroke();
  ctx.setLineDash([]);
}

// P6 — POSITION SIZING CALCULATOR
// ══════════════════════════════════════════
function buildSizingCalc() {
  const accountSize = 100000;
  const riskPct = 2;
  const premium = 9.30;

  function calc(acct, risk, prem) {
    const maxRisk = acct * (risk/100);
    const costPerContract = prem * 100;
    const recommended = Math.max(1, Math.floor(maxRisk / costPerContract));
    const kelly = Math.max(1, Math.floor((maxRisk * 0.25) / costPerContract));
    return {
      recommended,
      kelly,
      maxLoss: (recommended * costPerContract).toFixed(0),
      maxLossPct: (recommended * costPerContract / acct * 100).toFixed(1),
    };
  }

  set('tab-sizing', `
    <div style="font-size:15px;font-weight:700;margin-bottom:4px">📐 Position Sizing Calculator</div>
    <div style="font-size:11px;color:var(--text3);margin-bottom:16px">How many contracts should you trade? Enter your parameters below.</div>
    <div class="card card-body" style="margin-bottom:14px">
      <div style="display:grid;grid-template-columns:repeat(3,1fr);gap:14px;margin-bottom:14px">
        <div>
          <div class="form-label">Account Size ($)</div>
          <input type="number" id="sz-account" value="${accountSize}" oninput="refreshSizing()">
        </div>
        <div>
          <div class="form-label">Risk Per Trade (%)</div>
          <input type="number" id="sz-risk" value="${riskPct}" min="0.5" max="10" step="0.5" oninput="refreshSizing()">
          <div style="font-size:10px;color:var(--text3);margin-top:3px">Recommended: 1–2%</div>
        </div>
        <div>
          <div class="form-label">Premium Per Contract ($)</div>
          <input type="number" id="sz-premium" value="${premium}" step="0.10" oninput="refreshSizing()">
          <div style="font-size:10px;color:var(--text3);margin-top:3px">Per share (×100 = total cost)</div>
        </div>
      </div>
      <div id="sz-results" class="sizing-result">
        <div style="display:grid;grid-template-columns:repeat(2,1fr);gap:10px">
          ${[
            ['Recommended Contracts', calc(accountSize,riskPct,premium).recommended, 'cyan', 'Full Kelly adjusted for options risk'],
            ['Quarter Kelly Contracts', calc(accountSize,riskPct,premium).kelly, 'purple', 'Conservative Kelly Criterion sizing'],
            ['Max Loss ($)', '$'+calc(accountSize,riskPct,premium).maxLoss, 'red', 'If position goes to zero'],
            ['Max Loss (% of Account)', calc(accountSize,riskPct,premium).maxLossPct+'%', 'amber', 'Should stay under 2% per trade'],
          ].map(([l,v,c,desc]) => `
            <div style="padding:12px;background:var(--bg2);border-radius:var(--r);border:1px solid var(--border)">
              <div style="font-family:var(--mono);font-size:20px;font-weight:700;color:var(--${c})">${v}</div>
              <div style="font-size:10px;font-weight:700;text-transform:uppercase;letter-spacing:.1em;color:var(--text3);margin-top:3px">${l}</div>
              <div style="font-size:10px;color:var(--text3);margin-top:2px">${desc}</div>
            </div>`).join('')}
        </div>
        ${calc(accountSize,riskPct,premium).recommended > 5 ?
          `<div style="margin-top:10px;padding:8px 12px;background:var(--amber-bg);border-radius:var(--r);border:1px solid var(--amber-dim);font-size:11px;color:var(--amber)">⚠️ Position size is large. Consider reducing risk % or using fewer contracts.</div>` : ''}
      </div>
    </div>
    <div style="font-size:10px;color:var(--text3);text-align:center">⚠️ Educational tool only. Not financial advice.</div>
  `);
}

function refreshSizing() {
  const acct  = +($('sz-account')  && $('sz-account').value)  || 100000;
  const risk  = +($('sz-risk')     && $('sz-risk').value)     || 2;
  const prem  = +($('sz-premium')  && $('sz-premium').value)  || 9.30;
  const maxRisk = acct * (risk/100);
  const costPer = prem * 100;
  const rec  = Math.max(1, Math.floor(maxRisk / costPer));
  const kel  = Math.max(1, Math.floor((maxRisk * 0.25) / costPer));
  const ml   = (rec * costPer).toFixed(0);
  const mlp  = (rec * costPer / acct * 100).toFixed(1);
  const el = $('sz-results');
  if(!el) return;
  el.innerHTML = `
    <div style="display:grid;grid-template-columns:repeat(2,1fr);gap:10px">
      ${[
        ['Recommended Contracts', rec, 'cyan'],
        ['Quarter Kelly Contracts', kel, 'purple'],
        ['Max Loss ($)', '$'+ml, 'red'],
        ['Max Loss (% Account)', mlp+'%', 'amber'],
      ].map(([l,v,c]) => `
        <div style="padding:12px;background:var(--bg2);border-radius:var(--r);border:1px solid var(--border)">
          <div style="font-family:var(--mono);font-size:20px;font-weight:700;color:var(--${c})">${v}</div>

  `;
}

// P9 — OPTIONS FLOW HEATMAP
// ══════════════════════════════════════════
function buildHeatmap() {
  // Group MEGACAP_DATA by sector
  const sectorMap = {
    'Technology':    ['MSFT','AAPL','ORCL','GOOGL'],
    'Semiconductors':['NVDA','AMD','AVGO','SMCI'],
    'Consumer':      ['AMZN','META','NFLX','COST','WMT'],
    'Healthcare':    ['LLY','JNJ','UNH'],
    'Financials':    ['JPM','V'],
    'Energy':        ['XOM'],
  };

  const sectorStats = Object.entries(sectorMap).map(([sector, tickers]) => {
    const companies = tickers.map(t => MEGACAP_DATA.find(d => d.ticker===t)).filter(Boolean);
    const avgIVRank = companies.length ? Math.round(companies.reduce((s,c) => s+c.ivRank, 0)/companies.length) : 0;
    const bullish   = companies.filter(c => c.bias==='BULLISH').length;
    const bearish   = companies.filter(c => c.bias==='BEARISH').length;
    const netFlow   = bullish - bearish; // positive = more calls
    const totalPremium = companies.length * avgIVRank * 0.12; // simulated $M
    return { sector, companies, avgIVRank, bullish, bearish, netFlow, totalPremium: totalPremium.toFixed(1) };
  }).sort((a,b) => b.avgIVRank - a.avgIVRank);

  const cellColor = netFlow => {
    if(netFlow > 1)  return { bg:'rgba(0,212,170,0.15)', border:'var(--cyan-dim)',  text:'var(--cyan)',   label:'Call Heavy' };
    if(netFlow < -1) return { bg:'rgba(255,71,87,0.15)', border:'var(--red-dim)',   text:'var(--red)',    label:'Put Heavy'  };
    return              { bg:'rgba(139,127,255,0.1)',  border:'var(--purple-dim)', text:'var(--purple)', label:'Neutral'    };
  };

  set('tab-heatmap', `
    <div style="display:flex;align-items:center;justify-content:space-between;margin-bottom:14px;flex-wrap:wrap;gap:10px">
      <div>
        <div style="font-size:15px;font-weight:700">🔥 Options Flow Heatmap</div>
        <div style="font-size:11px;color:var(--text3);margin-top:2px">Sector-level call vs put flow · Mega-cap companies >$200B · Updated every 5 min</div>
      </div>
      <div style="display:flex;gap:8px;align-items:center">
        ${[['🟢 Call Heavy','cyan'],['⚪ Neutral','purple'],['🔴 Put Heavy','red']].map(([l,c]) =>
          `<span style="font-size:10px;color:var(--text2);display:flex;align-items:center;gap:4px"><span style="width:10px;height:10px;border-radius:2px;background:var(--${c}-bg);border:1px solid var(--${c}-dim);display:inline-block"></span>${l}</span>`
        ).join('')}
      </div>
    </div>

    <!-- Heatmap grid -->
    <div style="display:grid;grid-template-columns:repeat(3,1fr);gap:12px;margin-bottom:16px">
      ${sectorStats.map(s => {
        const col = cellColor(s.netFlow);
        return `
          <div class="heatmap-cell" style="background:${col.bg};border:1px solid ${col.border};padding:16px"
               onclick="showToast('${s.sector}','IV Rank ${s.avgIVRank} · ${s.bullish} bullish / ${s.bearish} bearish','${s.netFlow>0?'cyan':s.netFlow<0?'red':'purple'}')">
            <div style="display:flex;align-items:flex-start;justify-content:space-between;margin-bottom:8px">
              <div style="font-size:13px;font-weight:700">${s.sector}</div>
              <div style="font-family:var(--mono);font-size:18px;font-weight:800;color:${col.text}">${s.avgIVRank}</div>
            </div>
            <div style="font-size:10px;color:${col.text};font-weight:700;margin-bottom:6px">${col.label}</div>
            <div style="height:5px;background:var(--bg4);border-radius:3px;overflow:hidden;margin-bottom:8px">
              <div style="width:${s.avgIVRank}%;height:100%;background:${col.text};border-radius:3px;opacity:0.8"></div>
            </div>
            <div style="display:flex;justify-content:space-between;font-size:10px;color:var(--text3)">
              <span>${s.companies.length} companies</span>
              <span>~$${s.totalPremium}M flow</span>
            </div>
            <div style="display:flex;gap:6px;margin-top:8px;flex-wrap:wrap">
              ${s.companies.slice(0,4).map(c =>
                `<span style="font-family:var(--mono);font-size:9px;padding:2px 6px;border-radius:4px;background:var(--bg4);color:var(--text2)">${c.ticker}</span>`
              ).join('')}
